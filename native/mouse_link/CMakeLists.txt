cmake_minimum_required(VERSION 3.20)
project(mouse_link LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MOUSE_LISTENER_ONLY "Build the passive listener without Interception SDK" OFF)
add_library(mouse_link_settings INTERFACE)
target_compile_definitions(mouse_link_settings INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(mouse_link_settings INTERFACE /utf-8 /W4)
enable_testing()
add_executable(mouse_virtual_packet_tests virtual_mouse_packet_tests.cpp)
target_link_libraries(mouse_virtual_packet_tests PRIVATE mouse_link_settings)
add_test(NAME MouseVirtualPacketContract COMMAND mouse_virtual_packet_tests)
add_executable(mouse_virtual_relay virtual_mouse_relay.cpp fakerinput_output.cpp
    ../mouse_native/mouse_interception_transport.cpp)
target_include_directories(mouse_virtual_relay PRIVATE ..)
target_link_libraries(mouse_virtual_relay PRIVATE mouse_link_settings user32 hid setupapi cfgmgr32)
add_executable(mouse_virtual_output_probe virtual_mouse_output_probe.cpp fakerinput_output.cpp)
target_link_libraries(mouse_virtual_output_probe PRIVATE mouse_link_settings user32 hid setupapi cfgmgr32)
add_executable(mouse_existing_interception_tests ../mouse_native/mouse_interception_transport_tests.cpp
    ../mouse_native/mouse_interception_transport.cpp)
target_include_directories(mouse_existing_interception_tests PRIVATE ..)
target_link_libraries(mouse_existing_interception_tests PRIVATE mouse_link_settings user32)
add_test(NAME MouseExistingInterceptionContract COMMAND mouse_existing_interception_tests)
add_library(interception_query_fixture SHARED interception_query_fixture.cpp)
target_link_libraries(interception_query_fixture PRIVATE mouse_link_settings)
set(_query_fixture_directory "${CMAKE_CURRENT_BINARY_DIR}/filter-query-contract/$<CONFIG>")
set_target_properties(interception_query_fixture PROPERTIES OUTPUT_NAME interception
    RUNTIME_OUTPUT_DIRECTORY "${_query_fixture_directory}")
add_executable(interception_query_contract_tests interception_query_contract_tests.cpp
    ../mouse_native/mouse_interception_transport.cpp)
target_include_directories(interception_query_contract_tests PRIVATE ..)
target_link_libraries(interception_query_contract_tests PRIVATE mouse_link_settings user32)
set_target_properties(interception_query_contract_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${_query_fixture_directory}")
add_dependencies(interception_query_contract_tests interception_query_fixture)
add_test(NAME MouseInterceptionBrokenQueryRegression COMMAND interception_query_contract_tests)
add_test(NAME MouseExistingInterceptionLeakNegativeControl COMMAND mouse_existing_interception_tests --leak-fixture)
set_tests_properties(MouseExistingInterceptionLeakNegativeControl PROPERTIES WILL_FAIL TRUE)
set(_virtual_interception_dll "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/mouse_link/deps/Interception/library/x64/interception.dll")
if(EXISTS "${_virtual_interception_dll}")
    file(SHA256 "${_virtual_interception_dll}" _virtual_interception_sha)
    if(NOT _virtual_interception_sha STREQUAL "ab88164c11b1b48488772d4c3bfaa4509d5b0ae9dbc5a691dc4f96f0260443c8")
        message(FATAL_ERROR "Unexpected Interception DLL bytes")
    endif()
    add_custom_command(TARGET mouse_virtual_relay POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_virtual_interception_dll}"
        "$<TARGET_FILE_DIR:mouse_virtual_relay>/interception.dll")
endif()
add_test(NAME MouseVirtualLifecycleNoCapture COMMAND mouse_virtual_relay --smoke)
add_test(NAME MouseVirtualWatchdogNoCapture COMMAND mouse_virtual_relay --watchdog-test)
add_test(NAME MouseVirtualKernelExitPendingNegativeControl COMMAND mouse_virtual_relay --exit-pending-test)
set_tests_properties(MouseVirtualKernelExitPendingNegativeControl PROPERTIES
    WILL_FAIL TRUE RESOURCE_LOCK mouse_link_hotkeys TIMEOUT 12)
set_tests_properties(MouseVirtualLifecycleNoCapture MouseVirtualWatchdogNoCapture PROPERTIES
    RESOURCE_LOCK mouse_link_hotkeys TIMEOUT 12)
add_executable(mouse_input_monitor input_monitor.cpp)
target_link_libraries(mouse_input_monitor PRIVATE mouse_link_settings user32)
add_executable(mouse_input_provenance_tests input_provenance_tests.cpp)
target_link_libraries(mouse_input_provenance_tests PRIVATE mouse_link_settings)
add_test(NAME MouseInputProvenance COMMAND mouse_input_provenance_tests)
# SendInput experiments are explicit/manual, never part of unattended CTest.
if(MOUSE_LISTENER_ONLY)
    return()
endif()
set(INTERCEPTION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/mouse_link/deps/Interception" CACHE PATH "Official Interception v1.0.1 package")
if(NOT EXISTS "${INTERCEPTION_ROOT}/library/interception.h")
    message(FATAL_ERROR "Run scripts/verify/build_mouse_link.ps1 to fetch the pinned dependency first")
endif()
target_include_directories(mouse_link_settings INTERFACE "${INTERCEPTION_ROOT}/library")
add_executable(mouse_link_tool main.cpp)
target_link_libraries(mouse_link_tool PRIVATE mouse_link_settings
    "${INTERCEPTION_ROOT}/library/x64/interception.lib" user32 gdi32)
add_custom_command(TARGET mouse_link_tool POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${INTERCEPTION_ROOT}/library/x64/interception.dll" "$<TARGET_FILE_DIR:mouse_link_tool>/interception.dll")
add_executable(mouse_link_tests tests.cpp)
target_link_libraries(mouse_link_tests PRIVATE mouse_link_settings)
add_test(NAME MouseLinkPolicyAndOracle COMMAND mouse_link_tests)
add_test(NAME MouseLinkProcessesNoCapture COMMAND mouse_link_tool --smoke)
add_test(NAME MouseLinkIndependentWatchdog COMMAND mouse_link_tool --watchdog-test)
set_tests_properties(MouseLinkProcessesNoCapture MouseLinkIndependentWatchdog PROPERTIES
    RESOURCE_LOCK mouse_link_hotkeys TIMEOUT 15)
