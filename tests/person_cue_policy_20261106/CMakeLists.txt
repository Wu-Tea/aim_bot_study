cmake_minimum_required(VERSION 3.26)
project(person_cue_policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(VERSION IN ITEMS before after)
 if(VERSION STREQUAL "before")
  set(SOURCE "${ROOT}/runs/person-cue-policy-20261106/baseline/native")
 else()
  set(SOURCE "${ROOT}/native")
 endif()
 add_library(selector_${VERSION} STATIC "${SOURCE}/vision_native/src/target_selector.cpp" "${SOURCE}/vision_native/src/target_selector_fusion.cpp" "${SOURCE}/vision_native/src/cue_identity.cpp")
 if(VERSION STREQUAL "after")
  target_sources(selector_${VERSION} PRIVATE "${SOURCE}/vision_native/src/cue_acquisition_policy.cpp")
 endif()
 target_include_directories(selector_${VERSION} PUBLIC "${SOURCE}" "${SOURCE}/vision_native/include" "${ROOT}/native")
 target_compile_definitions(selector_${VERSION} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(selector_${VERSION} PUBLIC /utf-8)
 add_executable(cases_${VERSION} "${ROOT}/native/vision_native/src/person_cue_policy_tests.cpp")
 target_compile_definitions(cases_${VERSION} PRIVATE PERSON_CUE_STANDALONE)
 if(VERSION STREQUAL "before")
  target_compile_definitions(cases_${VERSION} PRIVATE PERSON_CUE_BASELINE)
 endif()
 target_link_libraries(cases_${VERSION} PRIVATE selector_${VERSION})
 add_executable(replay_${VERSION} "${ROOT}/tests/november_acquisition_20261102/replay.cpp")
 target_link_libraries(replay_${VERSION} PRIVATE selector_${VERSION})
endforeach()
add_executable(controller_cases EXCLUDE_FROM_ALL controller_main.cpp
 "${ROOT}/native/controller_native/target_pipeline_integration_tests.cpp")
target_include_directories(controller_cases PRIVATE "${ROOT}/native" "${ROOT}/native/vision_native/include"
 "${ROOT}/runs/person-cue-policy-20261106/runtime-build/generated")
target_compile_definitions(controller_cases PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(controller_cases PRIVATE /utf-8)
target_link_libraries(controller_cases PRIVATE
 "${ROOT}/runs/person-cue-policy-20261106/runtime-build/$<CONFIG>/controller_native_core.lib" windowsapp)
