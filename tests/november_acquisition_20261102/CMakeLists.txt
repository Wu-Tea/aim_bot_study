cmake_minimum_required(VERSION 3.26)
project(november_acquisition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OLD "${ROOT}/runs/incident-20261102/fix/baseline/native")
foreach(VERSION IN ITEMS before after)
 if(VERSION STREQUAL "before")
  set(SOURCE "${OLD}")
 else()
  set(SOURCE "${ROOT}/native")
 endif()
 add_library(selector_${VERSION} STATIC "${SOURCE}/vision_native/src/target_selector.cpp" "${SOURCE}/vision_native/src/target_selector_fusion.cpp" "${SOURCE}/vision_native/src/cue_identity.cpp")
 target_include_directories(selector_${VERSION} PUBLIC "${SOURCE}" "${SOURCE}/vision_native/include")
 target_compile_definitions(selector_${VERSION} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(selector_${VERSION} PUBLIC /utf-8)
 add_executable(cases_${VERSION} "${ROOT}/native/vision_native/src/november_acquisition_tests.cpp")
 target_compile_definitions(cases_${VERSION} PRIVATE NOVEMBER_STANDALONE)
 target_include_directories(cases_${VERSION} PRIVATE "${ROOT}/native")
 target_link_libraries(cases_${VERSION} PRIVATE selector_${VERSION})
 add_executable(replay_${VERSION} replay.cpp)
 target_link_libraries(replay_${VERSION} PRIVATE selector_${VERSION})
endforeach()
add_executable(color_probe color_probe.cpp)
target_link_libraries(color_probe PRIVATE selector_after)
