cmake_minimum_required(VERSION 3.26)
project(cue_offset_reproduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BASELINE_NATIVE "${ROOT}/runs/cue-offset-20261106/baseline/native" CACHE PATH "Frozen before-change native source")
foreach(VERSION IN ITEMS before after)
 if(VERSION STREQUAL "before")
  set(SOURCE "${BASELINE_NATIVE}")
 else()
  set(SOURCE "${ROOT}/native")
 endif()
 add_library(selector_${VERSION} STATIC
  "${SOURCE}/vision_native/src/target_selector.cpp"
  "${SOURCE}/vision_native/src/target_selector_fusion.cpp"
  "${SOURCE}/vision_native/src/cue_identity.cpp"
  "${SOURCE}/vision_native/src/cue_acquisition_policy.cpp")
 target_include_directories(selector_${VERSION} PUBLIC "${SOURCE}" "${SOURCE}/vision_native/include" "${ROOT}/native")
 target_compile_definitions(selector_${VERSION} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(selector_${VERSION} PUBLIC /utf-8)
 add_executable(probe_${VERSION} probe.cpp)
 target_link_libraries(probe_${VERSION} PRIVATE selector_${VERSION})
 add_executable(replay_${VERSION} "${ROOT}/tests/november_acquisition_20261102/replay.cpp")
 target_link_libraries(replay_${VERSION} PRIVATE selector_${VERSION})
 add_executable(material_${VERSION} material.cpp)
 target_link_libraries(material_${VERSION} PRIVATE selector_${VERSION})
 add_executable(cases_${VERSION} "${ROOT}/native/vision_native/src/cue_offset_tests.cpp")
 target_compile_definitions(cases_${VERSION} PRIVATE CUE_OFFSET_STANDALONE)
 target_link_libraries(cases_${VERSION} PRIVATE selector_${VERSION})
 if(VERSION STREQUAL "after")
  add_executable(november_after "${ROOT}/native/vision_native/src/november_acquisition_tests.cpp")
  target_compile_definitions(november_after PRIVATE NOVEMBER_STANDALONE)
  target_link_libraries(november_after PRIVATE selector_after)
  add_executable(visible_after "${ROOT}/native/vision_native/src/visible_selection_contract_tests.cpp")
  target_compile_definitions(visible_after PRIVATE VISIBLE_SELECTION_STANDALONE)
  target_link_libraries(visible_after PRIVATE selector_after)
 endif()
endforeach()
