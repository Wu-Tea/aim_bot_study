cmake_minimum_required(VERSION 3.26)
project(palette_study LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BASE "${ROOT}/runs/palette-20261023/reference")
add_executable(replay replay.cpp "${BASE}/reference.cpp" "${BASE}/reference_selector.cpp" "${BASE}/reference_selector_fusion.cpp" "${ROOT}/native/runtime_app/runtime_timing.cpp")
target_include_directories(replay PRIVATE "${BASE}" "${ROOT}/native/vision_native/include" "${ROOT}/native")
target_compile_definitions(replay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(replay PRIVATE /utf-8)
target_link_libraries(replay PRIVATE "${ROOT}/native/build/vision_native/Release/vision_native_core.lib" winmm)
add_executable(probe probe.cpp)
target_include_directories(probe PRIVATE "${ROOT}/native/vision_native/include" "${ROOT}/native")
target_compile_definitions(probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(probe PRIVATE "${ROOT}/native/build/vision_native/Release/vision_native_core.lib")
