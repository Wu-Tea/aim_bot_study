project(hipfire_flick_study LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(DEV "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BEFORE "D:/work/AI/yolo-study-001-clean" CACHE PATH "Unmodified baseline workspace")
configure_file("${DEV}/native/controller_native/incident_fixture_support.h"
    "${CMAKE_CURRENT_BINARY_DIR}/fixture/controller_native/incident_fixture_support.h" COPYONLY)
foreach(version before after)
  if(version STREQUAL "before")
    set(ROOT "${BEFORE}")
  else()

  endif()
  foreach(kind probe closed_loop scope_regression)
    if(kind STREQUAL "closed_loop")
      set(SOURCE "${DEV}/tests/ads_cooperation_20261019/closed_loop.cpp")
    elseif(kind STREQUAL "probe")
      set(SOURCE probe.cpp)
    else()
      set(SOURCE scope_regression.cpp)
    endif()
    add_executable(${kind}_${version} "${SOURCE}")
    target_include_directories(${kind}_${version} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture" "${ROOT}/native" "${ROOT}/native/controller_native" "${ROOT}/native/vision_native/include" "${ROOT}/native/build/generated")

    if(version STREQUAL "after")
      target_compile_definitions(${kind}_${version} PRIVATE STUDY_AFTER)
    endif()
    target_compile_options(${kind}_${version} PRIVATE /utf-8)
    target_link_libraries(${kind}_${version} PRIVATE "${ROOT}/native/build/Release/controller_native_core.lib" windowsapp)
  endforeach()
endforeach()
