cmake_minimum_required(VERSION 3.26)
project(ads_cooperation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(P "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
configure_file("${P}/native/controller_native/incident_fixture_support.h"
    "${CMAKE_CURRENT_BINARY_DIR}/fixture/controller_native/incident_fixture_support.h" COPYONLY)
foreach(version pre before after)
  if(version STREQUAL "after")
    set(root "${P}")
    set(lib "${P}/native/build/Release/controller_native_core.lib")
    set(gen "${root}/native/build/generated")
  else()
    set(root "${P}/runs/ads-cooperation-20261019/${version}")
    set(PROJECT_SOURCE_DIR "${root}/native")
    set(PROJECT_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/${version}")
    include("${root}/native/cmake/EditableParameters.cmake")
    set(gen "${PROJECT_BINARY_DIR}/generated")
    file(READ "${root}/native/controller_native/controller_targets.cmake" targets)
    string(REGEX MATCH "add_library\\(controller_native_core STATIC([^)]*)\\)" matched "${targets}")
    string(REGEX MATCHALL "[A-Za-z_./0-9-]+\\.cpp" sources "${CMAKE_MATCH_1}")
    list(TRANSFORM sources PREPEND "${root}/native/")
    add_library(core_${version} STATIC ${sources})
    target_include_directories(core_${version} PRIVATE "${gen}" "${root}/native" "${root}/native/vision_native/include")
    target_compile_definitions(core_${version} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(core_${version} PRIVATE /utf-8)
    set(lib core_${version})
  endif()
  foreach(kind comparison closed_loop live)
    add_executable(${kind}_${version} ${kind}.cpp)
    target_include_directories(${kind}_${version} PRIVATE "${root}/native" "${root}/native/controller_native" "${root}/native/vision_native/include" "${gen}" "${CMAKE_CURRENT_BINARY_DIR}/fixture")
    target_compile_definitions(${kind}_${version} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(${kind}_${version} PRIVATE /utf-8)
    if(version STREQUAL "after")
      target_compile_definitions(${kind}_${version} PRIVATE AFTER)
    endif()
    target_link_libraries(${kind}_${version} PRIVATE "${lib}" windowsapp)
  endforeach()
endforeach()
